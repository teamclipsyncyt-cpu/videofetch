function checkVideo() { const input = document.getElementById("videoUrl"); const message = document.getElementById("message"); const urlText = input.value.trim(); if (!urlText) { message.textContent = "Please paste a video link."; return; } let url; try { url = new URL(urlText); } catch { message.textContent = "Please enter a valid URL."; return; } if (!["https:", "http:"].includes(url.protocol)) { message.textContent = "Invalid video link."; return; } const host = url.hostname.toLowerCase(); // Platform links: open the original platform if ( host.includes("youtube.com") || host === "youtu.be" || host.includes("instagram.com") ) { message.textContent = "This platform does not allow unauthorized downloads. Open the original video to use its authorized download options."; window.open(url.href, "_blank", "noopener"); return; } // Direct video file links only const allowed = /\.(mp4|webm|mov|m4v)$/i.test(url.pathname); if (!allowed) { message.textContent = "Unsupported link. Please use a direct MP4, WebM or MOV video link."; return; } const link = document.createElement("a"); link.href = url.href; link.download = "VideoFetch-video"; link.target = "_blank"; link.rel = "noopener"; document.body.appendChild(link); link.click(); link.remove(); message.textContent = "Download requested. Your browser may open or save the video depending on the source server."; }
